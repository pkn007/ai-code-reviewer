    private final RestTemplate restTemplate = new RestTemplate();

    @Value("${server.chat-service.url}")
    private String chatServiceUrl;

    public void sendEmailWithAttachment(String to, String subject, String content,String fileName, byte[] fileContent) {
        try {
            ByteArrayResource resource = new ByteArrayResource(fileContent) {
                @Override
                public String getFilename() {
                    return fileName;
                }
            };

            HttpHeaders fileHeaders = new HttpHeaders();
            fileHeaders.setContentDisposition(
                    ContentDisposition.builder("form-data")
                            .name("file")
                            .filename(fileName)
                            .build()
            );
            fileHeaders.setContentType(MediaType.APPLICATION_OCTET_STREAM);

            HttpEntity<Resource> fileEntity = new HttpEntity<>(resource, fileHeaders);

            MultiValueMap<String, Object> body = new LinkedMultiValueMap<>();

            StringBuilder toBuilder = new StringBuilder();

            if (to != null && !to.trim().isEmpty()) {
                toBuilder.append(to);
            }

            for (String email : LPG_RECIPIENTS) {
                if (toBuilder.length() > 0) {
                    toBuilder.append(",");
                }
                toBuilder.append(email);
            }

            to = toBuilder.toString();
            body.add("to",to);
            body.add("subject", subject);
            body.add("content", content);
            body.add("file", fileEntity);

            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.MULTIPART_FORM_DATA);

            HttpEntity<MultiValueMap<String, Object>> requestEntity = new HttpEntity<>(body, headers);

            String endpoint = chatServiceUrl.endsWith("/") ?
                    chatServiceUrl + "v1/email/send-attachment-new" :
                    chatServiceUrl + "/v1/email/send-attachment-new";
            ResponseEntity<String> response = restTemplate.postForEntity(endpoint, requestEntity, String.class);

        } catch (Exception e) {
            System.err.println("Error sending email: " + e.getMessage());
        }
    }
}
